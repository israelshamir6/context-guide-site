`;
}

// ── PDF GENERATOR ──────────────────────────────────────────────────────
function generatePDF(sections, filename, docTitle) {
  loadJsPDF(() => {
    const {jsPDF} = window.jspdf;
    const doc = new jsPDF({orientation:'portrait', unit:'mm', format:'a4'});
    const W = doc.internal.pageSize.getWidth();
    const H = doc.internal.pageSize.getHeight();
    const ml=18, mr=18, mt=22, mb=22, cw=W-ml-mr;
    let y=mt;

    const C={
      gold:[218,165,32], dark:[13,17,23], navy:[26,58,92], steel:[44,95,138],
      body:[35,35,35],  grey:[150,150,150], cream:[255,252,230], amber:[180,83,9],
      lblGrey:[120,120,120]
    };

    function footer(){
      doc.setFontSize(7.5); doc.setFont('helvetica','normal');
      doc.setTextColor(...C.grey);
      doc.text('Forms of Context in Textual Analysis — Comprehensive Scholarly Guide', ml, H-10);
      doc.text(String(doc.internal.getNumberOfPages()), W-mr, H-10, {align:'right'});
      doc.setTextColor(...C.body);
    }
    function np(){ footer(); doc.addPage(); y=mt; }
    function chk(h){ if(y+h>H-mb) np(); }

    // Cover
    doc.setFillColor(...C.dark); doc.rect(0,0,W,H,'F');
    doc.setDrawColor(...C.gold); doc.setLineWidth(0.5);
    doc.line(ml,32,W-mr,32);
    doc.setFontSize(8.5); doc.setFont('helvetica','bold'); doc.setTextColor(...C.gold);
    doc.text('FORMS OF CONTEXT IN TEXTUAL ANALYSIS',W/2,28,{align:'center'});
    doc.setFontSize(18); doc.setFont('helvetica','bold'); doc.setTextColor(255,255,255);
    const tls=doc.splitTextToSize(docTitle,cw-10);
    let ty=50; tls.forEach(l=>{doc.text(l,W/2,ty,{align:'center'});ty+=11;});
    doc.setFontSize(8.5); doc.setFont('helvetica','normal'); doc.setTextColor(...C.grey);
    doc.text('A Comprehensive Scholarly Guide',W/2,ty+8,{align:'center'});
    doc.line(ml,H-32,W-mr,H-32);
    doc.text('formsofcontext.com',W/2,H-22,{align:'center'});
    doc.addPage(); y=mt; doc.setTextColor(...C.body);

    // Render each section
    for(const {title, body} of sections){
      // Section heading bar
      chk(20); if(y>mt+4) y+=6;
      doc.setFillColor(...C.dark); doc.rect(ml-3,y-7,cw+6,13,'F');
      doc.setFontSize(12.5); doc.setFont('helvetica','bold'); doc.setTextColor(...C.gold);
      const hls=doc.splitTextToSize(title,cw-4);
      doc.text(hls,ml,y); y+=hls.length*7+8; doc.setTextColor(...C.body);

      // Process body text line by line
      const lines=body.split('\n');
      for(const raw of lines){
        const l=raw.trim();
        if(!l){ y+=1.5; continue; }

        // Section label [IN BRACKETS]
        if(/^\[.{3,60}\]$/.test(l)){
          chk(10); y+=3;
          doc.setDrawColor(210,210,210); doc.setLineWidth(0.3);
          doc.line(ml,y-1,W-mr,y-1);
          doc.setFontSize(7.5); doc.setFont('helvetica','bold');
          doc.setTextColor(...C.lblGrey);
          doc.text(l.replace(/^\[|\]$/g,'').toUpperCase(),ml,y);
          y+=5.5; doc.setTextColor(...C.body);
          continue;
        }

        // Key quote "..."
        if(l.startsWith('"') && l.length>10){
          doc.setFontSize(9); doc.setFont('helvetica','italic'); doc.setTextColor(60,50,30);
          const ls=doc.splitTextToSize(l,cw-10);
          const bh=ls.length*5+10;
          chk(bh);
          doc.setFillColor(...C.cream); doc.rect(ml,y-4,cw,bh,'F');
          doc.setFillColor(...C.amber); doc.rect(ml,y-4,2,bh,'F');
          doc.text(ls,ml+5,y); y+=bh+3;
          doc.setFont('helvetica','normal'); doc.setTextColor(...C.body);
          continue;
        }

        // Attribution line starting with —
        if(l.startsWith('—')){
          doc.setFontSize(8.5); doc.setFont('helvetica','italic');
          doc.setTextColor(100,100,100);
          const ls=doc.splitTextToSize(l,cw-8);
          chk(ls.length*5+2);
          doc.text(ls,ml+6,y); y+=ls.length*5+3;
          doc.setFont('helvetica','normal'); doc.setTextColor(...C.body);
          continue;
        }

        // Case study lines
        if(l.startsWith('Case Study')||l.startsWith('Source:')||l.startsWith('Analysis:')){
          doc.setFontSize(9); doc.setFont('helvetica','normal');
          doc.setTextColor(30,60,120);
          const ls=doc.splitTextToSize(l,cw-8);
          chk(ls.length*5+2);
          doc.text(ls,ml+4,y); y+=ls.length*5+2;
          doc.setTextColor(...C.body);
          continue;
        }

        // Indented citation (scholarly source)
        if(raw.startsWith('  ')&&(l.includes('(1')||l.includes('(2'))&&l.length>20){
          doc.setFontSize(8.5); doc.setFont('helvetica','bold');
          doc.setTextColor(...C.navy);
          const ls=doc.splitTextToSize(l,cw-8);
          chk(ls.length*5+2);
          doc.text(ls,ml+4,y); y+=ls.length*5+2;
          doc.setFont('helvetica','normal'); doc.setTextColor(...C.body);
          continue;
        }

        // Normal paragraph
        doc.setFontSize(9.5); doc.setFont('helvetica','normal'); doc.setTextColor(...C.body);
        const ls=doc.splitTextToSize(l,cw);
        for(const line of ls){ chk(5); doc.text(line,ml,y); y+=5; }
        y+=1;
      }
      // Section divider
      chk(10); y+=4;
      doc.setDrawColor(200,200,200); doc.setLineWidth(0.4);
      doc.line(ml,y,W-mr,y); y+=7;
    }
    footer();
    doc.save(filename+'.pdf');
  });
}

// ── BLOB DOWNLOAD ──────────────────────────────────────────────────────
function setProgress(id,msg){
  const el=document.getElementById(id);
  if(el){el.textContent=msg;el.classList.toggle('show',!!msg);}
}
function triggerBlob(content,filename,mime){
  const blob=new Blob([content],{type:mime});
  const url=URL.createObjectURL(blob);
  const a=document.createElement('a');
  a.href=url; a.download=filename;
  document.body.appendChild(a); a.click();
  setTimeout(()=>{URL.revokeObjectURL(url);a.remove();},1500);
}
function dispatch(sections,baseFilename,docTitle,fmt,progId){
  if(fmt==='pdf'){
    generatePDF(sections,baseFilename,docTitle);
    setProgress(progId,'✅ PDF generated — check your Downloads folder.');
  } else if(fmt==='word'){
    triggerBlob(wordDoc(sections,docTitle),baseFilename+'.doc','application/msword');
    setProgress(progId,'✅ Word doc downloaded!');
  } else {
    triggerBlob(textDoc(sections),baseFilename+'.txt','text/plain');
    setProgress(progId,'✅ Text file downloaded!');
  }
  setTimeout(()=>setProgress(progId,''),6000);
}

// AUDIO ENGINE v5 — sequential chunks, done-guard, no double-fallback
// ═══════════════════════════════════════════════════════
window.TTS = (function() {
  'use strict';
  let activePlayBtn = null;
  let activeBar     = null;
  let isPaused      = false;
  let currentBrianAudio = null;
  let currentBrianQueue = [];
  let ttsGen = 0; // incremented on every new speak() — invalidates stale callbacks

  /* ══════════════════════════════════════════════════════
     TEXT NORMALISER
  ══════════════════════════════════════════════════════ */
  function cleanText(raw) {
    return raw
      .replace(/[Ͱ-Ͽἀ-῿]+/g,'')
      .replace(/[֐-׿]+/g,'')
      .replace(/[̀-ͯ]/g,'')
      .replace(/[āĀ]/g,'a').replace(/[ēĒ]/g,'e').replace(/[īĪ]/g,'i')
      .replace(/[ōŌ]/g,'o').replace(/[ūŪ]/g,'u')
      .replace(/[çÇ]/g,'s').replace(/[ṭ]/g,'t').replace(/[ḥ]/g,'h')
      .replace(/[šŠ]/g,'sh').replace(/[ẓ]/g,'z').replace(/[ḏ]/g,'d')
      .replace(/[▶⏸■🔈🔊▼▲]/g,'')
      .replace(/[\uD83C-􏰀-\uDFFF]/g,'')
      .replace(/[←-➿]/g,'')
      .replace(/&amp;/g,'and').replace(/&lt;/g,'less than').replace(/&gt;/g,'greater than')
      .replace(/&nbsp;/g,' ').replace(/&[a-z]+;/g,'')
      .replace(/https?:\/\/[^\s]+/g,'')
      .replace(/\[.*?\]/g,'')
      .replace(/\(cf\.\s*/gi,'see ').replace(/\(ibid\.\)/gi,'same source')
      .replace(/\be\.g\./gi,'for example').replace(/\bi\.e\./gi,'that is')
      .replace(/\bvs?\./gi,'versus').replace(/\bca\./gi,'around').replace(/\bpp?\./gi,'page')
      .replace(/\besp\./gi,'especially').replace(/\blit\./gi,'literally')
      .replace(/([A-Z])\.([A-Z])\./g,'$1 $2')
      .replace(/(\w)--(\w)/g,'$1, $2').replace(/(\w)—(\w)/g,'$1, $2')
      .replace(/(\w)–(\w)/g,'$1 to $2')
      .replace(/\s*[|]\s*/g,'. ')
      .replace(/\n{2,}/g,'. ').replace(/\n/g,' ')
      .replace(/ +([.!?,;:])/g,'$1')
      .replace(/([.!?])\s*/g,'$1 ')
      .replace(/ {2,}/g,' ')
      .trim();
  }

  /* ══════════════════════════════════════════════════════
     TEXT EXTRACTION — strips ALL UI chrome, audio buttons
  ══════════════════════════════════════════════════════ */
  function getTextOf(el) {
    const clone = el.cloneNode(true);
    clone.querySelectorAll(
      'img,canvas,svg,picture,video,audio,'+
      '.audio-bar,.ab-mini,.ab-ham,.ab-stop-btn,.ab-page-bar,'+
      'button,script,style,nav,'+
      '.ham-panel,.dl-modal,.cat-tabs,'+
      '.kjv-subtab-bar,.sub-tab-row,.tab-row,'+
      '[aria-hidden="true"]'
    ).forEach(x=>x.remove());
    const raw=(clone.innerText||clone.textContent||'').replace(/\s{3,}/g,'\n\n').trim();
    return cleanText(raw);
  }

  function getHeadingBlock(h) {
    const lvl=parseInt(h.tagName[1]);
    const clone=h.cloneNode(true);
    clone.querySelectorAll('.ab-mini,.ab-ham,button').forEach(x=>x.remove());
    let text=(clone.innerText||'')+'. ';
    let sib=h.nextElementSibling;
    while(sib){
      const tag=sib.tagName||'';
      if(/^H[1-6]$/.test(tag)&&parseInt(tag[1])<=lvl)break;
      if(!sib.classList.contains('audio-bar')&&!sib.classList.contains('ab-page-bar')){
        const c=sib.cloneNode(true);
        c.querySelectorAll('img,canvas,svg,.audio-bar,.ab-mini,.ab-page-bar,button,script').forEach(x=>x.remove());
        text+=(c.innerText||c.textContent||'')+' ';
      }
      sib=sib.nextElementSibling;
    }
    return cleanText(text.replace(/\s{3,}/g,' ').trim());
  }

  /* ══════════════════════════════════════════════════════
     PLAYBACK UI
  ══════════════════════════════════════════════════════ */
  function stopUI(){
    if(activePlayBtn){
      if(activePlayBtn.classList.contains('ab-play-btn'))activePlayBtn.innerHTML='▶ Read';
      else if(activePlayBtn.classList.contains('ab-page-btn'))activePlayBtn.innerHTML='▶ Read Page';
      else activePlayBtn.textContent='🔈';
      activePlayBtn.classList.remove('ab-active','playing');
    }
    if(activeBar){
      const w=activeBar.querySelector('.ab-waves');
      const s=activeBar.querySelector('.ab-stop-btn');
      if(w)w.classList.remove('ab-vis');
      if(s)s.classList.remove('ab-vis');
    }
    activePlayBtn=null;activeBar=null;
    const gb=document.getElementById('tts-btn');
    if(gb){gb.textContent='🔈';gb.classList.remove('playing');}
  }

  function stopAll(){
    ttsGen++; // ← KEY FIX: invalidates every pending Brian/WSA callback
    if(currentBrianAudio){
      try{currentBrianAudio.pause();}catch(e){}
      currentBrianAudio=null;
    }
    // pause only — no src='' (that triggers onerror which restarted old speech)
    currentBrianQueue.forEach(a=>{try{a.pause();}catch(e){}});
    currentBrianQueue=[];
    if(window.speechSynthesis)speechSynthesis.cancel();
    isPaused=false;
    stopUI();
  }

  function startUI(btn,bar){
    stopAll(); // ttsGen is now incremented
    activePlayBtn=btn;activeBar=bar;
    if(btn){
      if(btn.classList.contains('ab-play-btn')||btn.classList.contains('ab-page-btn'))
        btn.innerHTML='⏸ Pause';
      else{btn.textContent='⏸';btn.classList.add('playing');}
      btn.classList.add('ab-active');
    }
    if(bar){
      const w=bar.querySelector('.ab-waves');
      const s=bar.querySelector('.ab-stop-btn');
      if(w)w.classList.add('ab-vis');
      if(s)s.classList.add('ab-vis');
    }
  }

  /* ══════════════════════════════════════════════════════
     WEB SPEECH API — UK Male voice (fallback)
  ══════════════════════════════════════════════════════ */
  (function warmVoices(){
    if(!window.speechSynthesis)return;
    const v=speechSynthesis.getVoices();
    if(!v.length)speechSynthesis.onvoiceschanged=function(){speechSynthesis.getVoices();};
  })();

  const MALE_UK_PREF=[
    'Google UK English Male','Microsoft Ryan Online (Natural)',
    'Microsoft George Online (Natural)','Microsoft George',
    'Daniel','Arthur','Oliver','Harry','James',
    'Microsoft David','Microsoft Mark',
  ];

  function getBestMaleVoice(){
    const voices=window.speechSynthesis?window.speechSynthesis.getVoices():[];
    for(const n of MALE_UK_PREF){const v=voices.find(x=>x.name===n);if(v)return v;}
    const gb=voices.filter(x=>x.lang==='en-GB'||x.lang.startsWith('en-GB'));
    const gbM=gb.find(x=>/ryan|george|daniel|arthur|oliver|harry|james|male/i.test(x.name));
    if(gbM)return gbM;
    if(gb.length)return gb[0];
    return voices.find(x=>/david|mark|ryan|george|daniel|arthur|oliver|harry|james|male/i.test(x.name))||null;
  }

  function speakWSA(text,myGen,onEnd){
    const synth=window.speechSynthesis;
    if(!synth||ttsGen!==myGen){if(ttsGen===myGen)onEnd();return;}
    if(synth.speaking||synth.pending)synth.cancel();
    const MAX=460;
    const sentences=text.replace(/([.!?])\s+/g,'$1\n').split('\n').map(s=>s.trim()).filter(Boolean);
    const chunks=[];let buf='';
    for(const s of sentences){
      const c=(buf?buf+' ':'')+s;
      if(c.length>MAX&&buf){chunks.push(buf);buf=s;}else buf=c;
    }
    if(buf)chunks.push(buf);
    if(!chunks.length){if(ttsGen===myGen)onEnd();return;}
    let idx=0;
    const voice=getBestMaleVoice();
    function next(){
      if(ttsGen!==myGen)return; // generation guard
      if(idx>=chunks.length){onEnd();return;}
      const utt=new SpeechSynthesisUtterance(chunks[idx++]);
      utt.lang='en-GB';utt.rate=0.90;utt.pitch=0.88;utt.volume=1;
      if(voice)utt.voice=voice;
      utt.onend=next;
      utt.onerror=function(){if(ttsGen===myGen&&idx<chunks.length)next();else if(ttsGen===myGen)onEnd();};
      synth.speak(utt);
    }
    if(synth.getVoices().length>0){next();}
    else{
      let done=false;
      function go(){if(done||ttsGen!==myGen)return;done=true;synth.onvoiceschanged=null;clearInterval(poll);next();}
      synth.onvoiceschanged=go;
      const poll=setInterval(function(){if(synth.getVoices().length>0)go();},80);
      setTimeout(function(){clearInterval(poll);if(!done&&ttsGen===myGen){done=true;next();}},3000);
    }
  }

  /* ══════════════════════════════════════════════════════
     STREAMELEMENTS BRIAN (Amazon Polly UK Male)
     Preloads ALL chunks immediately — minimal inter-chunk gap
     Generation-safe: every callback checks ttsGen===myGen
  ══════════════════════════════════════════════════════ */
  const BRIAN_BASE='https://api.streamelements.com/kappa/v2/speech?voice=Brian&text=';
  const BRIAN_MAX=190;

  function chunkText(text,maxLen){
    const sentences=text.replace(/([.!?])\s+/g,'$1\n').split('\n').map(s=>s.trim()).filter(Boolean);
    const chunks=[];let buf='';
    for(const s of sentences){
      const candidate=(buf?buf+' ':'')+s;
      if(candidate.length<=maxLen){buf=candidate;}
      else{
        if(buf)chunks.push(buf);
        if(s.length>maxLen){
          const parts=s.split(/(?<=[,;])\s+/);
          let sub='';
          for(const p of parts){
            const c2=(sub?sub+' ':'')+p;
            if(c2.length<=maxLen){sub=c2;}
            else{if(sub)chunks.push(sub);sub=p;}
          }
          buf=sub;
        }else{buf=s;}
      }
    }
    if(buf)chunks.push(buf);
    return chunks;
  }

  function speakBrian(text,myGen,onEnd){
    if(ttsGen!==myGen)return;
    const chunks=chunkText(text,BRIAN_MAX);
    if(!chunks.length){if(ttsGen===myGen)onEnd();return;}
    // Sequential: one chunk at a time — avoids N×2 onerror+catch pile-up
    let idx=0;
    function playOne(){
      if(ttsGen!==myGen)return;
      if(idx>=chunks.length){onEnd();return;}
      const chunk=chunks[idx++];
      const audio=new Audio(BRIAN_BASE+encodeURIComponent(chunk));
      audio.preload='auto';
      currentBrianAudio=audio;
      currentBrianQueue=[audio];
      // done flag: only ONE of (onended / onerror / play().catch) runs — never both
      let done=false;
      function next(){
        if(done)return; done=true;
        if(ttsGen!==myGen)return;
        currentBrianAudio=null; currentBrianQueue=[];
        playOne();
      }
      function fail(){
        if(done)return; done=true;
        if(ttsGen!==myGen)return;
        currentBrianAudio=null; currentBrianQueue=[];
        // Fall back to Web Speech API for this chunk and all remaining
        speakWSA(chunks.slice(idx-1).join(' '),myGen,onEnd);
      }
      audio.onended=next;
      audio.onerror=fail;
      const p=audio.play();
      if(p&&p.catch)p.catch(fail);
    }
    playOne();
  }

  /* ══════════════════════════════════════════════════════
     MAIN speak() — Brian primary, WSA fallback
  ══════════════════════════════════════════════════════ */
  function speak(text,btn,bar){
    if(!text||!text.trim())return;

    // Same button + paused → RESUME
    if(activePlayBtn===btn&&isPaused){
      if(currentBrianAudio)currentBrianAudio.play().catch(function(){});
      else if(window.speechSynthesis&&speechSynthesis.paused)speechSynthesis.resume();
      isPaused=false;
      if(btn){
        if(btn.classList.contains('ab-play-btn')||btn.classList.contains('ab-page-btn'))
          btn.innerHTML='⏸ Pause';
        else{btn.textContent='⏸';btn.classList.add('playing');}
      }
      if(activeBar){const w=activeBar.querySelector('.ab-waves');if(w)w.classList.add('ab-vis');}
      return;
    }

    // Same button + playing → PAUSE
    if(activePlayBtn===btn&&!isPaused){
      if(currentBrianAudio)currentBrianAudio.pause();
      else if(window.speechSynthesis)speechSynthesis.pause();
      isPaused=true;
      if(btn){
        if(btn.classList.contains('ab-play-btn')||btn.classList.contains('ab-page-btn'))
          btn.innerHTML='▶ Resume';
        else btn.textContent='▶';
      }
      if(activeBar){const w=activeBar.querySelector('.ab-waves');if(w)w.classList.remove('ab-vis');}
      return;
    }

    // New button → stop everything, start fresh
    startUI(btn,bar);
    const myGen=ttsGen; // capture current generation
    const onEnd=()=>{
      if(ttsGen!==myGen)return; // stale — newer session running
      isPaused=false;currentBrianAudio=null;currentBrianQueue=[];
      stopUI();
    };
    try{speakBrian(text,myGen,onEnd);}
    catch(e){speakWSA(text,myGen,onEnd);}
  }

  /* ══════════════════════════════════════════════════════
     INJECTION — attaches audio buttons to every element type
     Button hierarchy:
       🔈 card header (mini)       — reads card
       ▶ Play All (group bar)     — reads ctx-card body
       ▶ Read (section bar)       — reads ctx-section / card-sm / box
       ▶ Read Page (page bar)     — reads full sub-section / panel
       🔈 heading mini             — reads heading + following siblings
       🔈 ham-item                 — reads menu item text
       🔈 floating #tts-btn        — reads entire active page
  ══════════════════════════════════════════════════════ */
  function SKIP(el){
    return el.closest('#topnav,.ham-panel,.dl-modal,.audio-bar,.ab-page-bar,#tts-btn,#tts-tooltip');
  }

  function injectEl(root){

    /* ── 0. Page-level: "Read Page" at top of every sub-section/panel ── */
    root.querySelectorAll('.sub-section,.sub-panel,.cat-panel,.content-panel,.kjv-panel').forEach(panel=>{
      if(panel.dataset.abPage)return;
      panel.dataset.abPage='1';
      const h=panel.querySelector('h2,h3,h4,.section-label,.ctx-section-label');
      const lbl=h?(h.innerText||'').trim().replace(/🔈/g,'').slice(0,60):'Read full page';
      panel.insertBefore(makePageBar(panel,lbl),panel.firstChild);
    });

    /* ── 1. ctx-card: mini on header + Play All on body ── */
    root.querySelectorAll('.ctx-card').forEach(card=>{
      if(card.dataset.ab)return;
      card.dataset.ab='1';
      const header=card.querySelector('.ctx-card-header');
      const chevron=card.querySelector('.ctx-chevron');
      const body=card.querySelector('.ctx-body');
      if(header&&!header.querySelector('.ab-mini')){
        const mini=makeMini(()=>getTextOf(card));
        mini.title='Read entire card';
        mini.style.cssText='font-size:14px;opacity:.45;margin-left:auto;margin-right:8px;flex-shrink:0';
        if(chevron)header.insertBefore(mini,chevron);
        else header.appendChild(mini);
      }
      if(body&&!body.querySelector(':scope > .audio-bar')){
        const h4=card.querySelector('h4');
        const lbl=h4?(h4.innerText||'').trim():'Read all subsections';
        body.insertBefore(makeGroupBar(body,lbl),body.firstChild);
      }
    });

    /* ── 2. ctx-section ── */
    root.querySelectorAll('.ctx-section').forEach(el=>{
      if(el.dataset.ab||SKIP(el))return;
      el.dataset.ab='1';
      const lbl=(el.querySelector('.ctx-section-label,.ctx-section-title')||{}).innerText||'Section';
      el.appendChild(makeBar(el,lbl));
    });

    /* ── 3. card-sm ── */
    root.querySelectorAll('.card-sm').forEach(el=>{
      if(el.dataset.ab||SKIP(el))return;
      el.dataset.ab='1';
      const lbl=(el.querySelector('h4,h5,.ctx-section-label,.card-title')||{}).innerText||'';
      el.appendChild(makeBar(el,lbl));
    });

    /* ── 4. Inline styled boxes (case study, highlight, vb) ── */
    root.querySelectorAll('.ctx-body > div[style],.ctx-section > div[style]').forEach(el=>{
      if(el.dataset.ab||SKIP(el))return;
      if(el.classList.contains('ctx-section')||el.classList.contains('ctx-grid'))return;
      if(el.querySelector('.ctx-card,.audio-bar'))return;
      const s=el.getAttribute('style')||'';
      if(!s.includes('border')&&!s.includes('background'))return;
      if((el.innerText||'').trim().length<20)return;
      el.dataset.ab='1';
      const lbl=(el.querySelector('[style*="uppercase"],[style*="text-transform"]')||
                 el.querySelector('[style*="font-weight:700"],[style*="font-weight:600"]')||
                 {}).innerText||'';
      el.appendChild(makeBar(el,(lbl||'Block').slice(0,52)));
    });

    /* ── 5. example-box / misapp-box ── */
    root.querySelectorAll('.example-box,.misapp-box').forEach(el=>{
      if(el.dataset.ab||SKIP(el))return;
      el.dataset.ab='1';
      const lbl=(el.querySelector('.ex-label,.mis-label')||{}).innerText||'Example';
      el.appendChild(makeBar(el,lbl));
    });

    /* ── 6. scholar-cite ── */
    root.querySelectorAll('.scholar-cite').forEach(el=>{
      if(el.dataset.ab||SKIP(el))return;
      el.dataset.ab='1';
      el.appendChild(makeBar(el,'Scholar'));
    });

    /* ── 7. omit-row ── */
    root.querySelectorAll('.omit-row').forEach(el=>{
      if(el.dataset.ab||SKIP(el))return;
      el.dataset.ab='1';
      el.appendChild(makeBar(el,''));
    });

    /* ── 8. Named card classes ── */
    const CARD_SEL=[
      '.card','.lexicon-card','.scholar-card','.home-card','.lex-card','.glos-card',
      '.principle-card','.error-card','.argument-card','.stat-box','.verdict-box',
      '.scripture-block','.quote-block','.cat-header-block','[class$="-card"]'
    ].join(',');

      if(el.closest('.ctx-body')&&!el.matches('.quote-block,.scripture-block,.argument-card'))return;
      el.dataset.ab='1';

    root.querySelectorAll('h2,h3,h4').forEach(h=>{

    });

    /* ── 11. h5 — inline mini speaker ── */
    root.querySelectorAll('h5').forEach(h=>{
      if(h.dataset.ab||SKIP(h))return;
      if(h.closest('.audio-bar,.cat-tabs,.ab-page-bar'))return;
      h.dataset.ab='1';
      h.appendChild(makeMini(()=>getHeadingBlock(h)));

    root.querySelectorAll('.page > .container > div,.page > .container > section').forEach(div=>{

      if(div.matches(CARD_SEL)||div.classList.contains('cat-panel')||

      item.dataset.ab='1';

  }

  /* ══════════════════════════════════════════════════════
     FULL-SITE INJECTION
  ══════════════════════════════════════════════════════ */
  function inject(){
    document.querySelectorAll('.audio-bar:not([data-static]),.ab-mini,.ab-ham,.ab-page-bar').forEach(el=>el.remove());
    document.querySelectorAll('[data-ab],[data-ab-page]').forEach(el=>{
      delete el.dataset.ab;delete el.dataset.abPage;
    });
    injectEl(document);
  }

  /* ══════════════════════════════════════════════════════
     GLOBAL FLOATING BUTTON (read entire active page)
  ══════════════════════════════════════════════════════ */
  function initGlobalBtn(){
    const btn=document.getElementById('tts-btn');
    if(!btn||btn.dataset.ab)return;
    btn.dataset.ab='1';btn.textContent='🔈';
    let dragging=false,moved=false,sx,sy,sl,st;
    function sd(cx,cy){const r=btn.getBoundingClientRect();sx=cx;sy=cy;sl=r.left;st=r.top;btn.style.right='auto';btn.style.bottom='auto';btn.style.left=sl+'px';btn.style.top=st+'px';dragging=true;moved=false;}
    function dd(cx,cy){if(!dragging)return;const dx=cx-sx,dy=cy-sy;if(Math.abs(dx)>4||Math.abs(dy)>4)moved=true;btn.style.left=Math.max(8,Math.min(innerWidth-btn.offsetWidth-8,sl+dx))+'px';btn.style.top=Math.max(8,Math.min(innerHeight-btn.offsetHeight-8,st+dy))+'px';}
    function ed(){if(!moved){const page=document.querySelector('.page.active,.sub-section.active');const text=page?getTextOf(page):'';if(activePlayBtn===btn){stopAll();}else{speak(text,btn,null);btn.textContent='🔊';btn.classList.add('playing');}}dragging=false;moved=false;}
    btn.addEventListener('mousedown',e=>{e.preventDefault();sd(e.clientX,e.clientY);});
    document.addEventListener('mousemove',e=>{if(dragging)dd(e.clientX,e.clientY);});
    document.addEventListener('mouseup',()=>{if(dragging)ed();});
    btn.addEventListener('touchstart',e=>{const t=e.touches[0];sd(t.clientX,t.clientY);},{passive:true});
    document.addEventListener('touchmove',e=>{if(!dragging)return;dd(e.touches[0].clientX,e.touches[0].clientY);},{passive:true});
    document.addEventListener('touchend',()=>{if(dragging)ed();});
  }

  document.addEventListener('DOMContentLoaded',()=>{initGlobalBtn();inject();});
  if(document.readyState!=='loading')setTimeout(()=>{initGlobalBtn();inject();},80);

  return { stopAll, inject, injectEl };
})();
